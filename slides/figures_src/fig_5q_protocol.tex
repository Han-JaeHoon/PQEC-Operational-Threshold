\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

\def\ya{4.0}   % ancilla a
\def\yb{3.0}   % A1
\def\yc{2.0}   % B1
\def\yd{1.0}   % A2
\def\ye{0.0}   % B2

\def\xl{1.35}
\def\xr{6.85}

\qwire{\ya}{\xl}{\xr}
\qwire{\yb}{\xl}{\xr}
\qwire{\yc}{\xl}{\xr}
\qwire{\yd}{\xl}{6.18}
\qwire{\ye}{\xl}{6.18}

% ---- inputs --------------------------------------------------------------
\node at (0.72,\ya) {\large $a:\ \ket{0}$};
\qbrace{0.42}{\yb+0.42}{\yc-0.42}{\rho}
\qbrace{0.42}{\yd+0.42}{\ye-0.42}{\rho}
\node at (0.96,\yb) {$A_1$};
\node at (0.96,\yc) {$B_1$};
\node at (0.96,\yd) {$A_2$};
\node at (0.96,\ye) {$B_2$};

% ---- the SWAP test -------------------------------------------------------
\qgate{2.00}{\ya}{H}
\qlink{3.10}{\ya}{\yd}\qctrl{3.10}{\ya}\qswap{3.10}{\yb}\qswap{3.10}{\yd}
\qlink{4.30}{\ya}{\ye}\qctrl{4.30}{\ya}\qswap{4.30}{\yc}\qswap{4.30}{\ye}
\qgate{5.40}{\ya}{H}

% ---- read-out ------------------------------------------------------------
\qmeter{6.45}{\ya}
\qmeterbig{6.45}{\yb}{\yc}{}
\qtrash{6.45}{\yd}
\qtrash{6.45}{\ye}

\node[right=8pt,align=left] at (\xr,\ya)
  {\lblsm{measure $Z_a$}\\[1pt]\note{\footnotesize parity weight}};
\node[right=8pt,align=left] at (\xr,{(\yb+\yc)/2})
  {\lblsm{measure the intended}\\[1pt]\lblsm{observable $O$ on $(A_1,B_1)$}};
\node[right=8pt,align=left] at (\xr,{(\yd+\ye)/2}) {\lblsm{discarded}};

% ---- the estimator -------------------------------------------------------
\node[anchor=west] at (0.42,-1.20)
  {$\displaystyle \avg{O}_{\mathcal{P}}
     =\frac{\avg{Z_a\otimes O_{A_1B_1}}}{\avg{Z_a\otimes I}}
     =\frac{\operatorname{Tr}(O\rho^{2})}{\operatorname{Tr}\rho^{2}}$};
\node[anchor=west,align=left] at (0.42,-2.20)
  {\note{Every shot is kept and weighted by the ancilla parity;
         the state $\rho^{2}/\operatorname{Tr}\rho^{2}$ is never prepared.}};

\end{tikzpicture}
